#include <WiFi.h>            // Bibliothèque pour gérer la connexion Wi-Fi
#include <PubSubClient.h>    // Bibliothèque pour MQTT
#include <ArduinoJson.h>     // Bibliothèque pour manipuler JSON
#include <ESP32Servo.h>      // Bibliothèque pour contrôler le servomoteur
#include <Wire.h>            // Bibliothèque I2C
#include <Adafruit_BME280.h> // Bibliothèque capteur BME280 (température, humidité, pression)

// ===== CONFIG =====
#define LED_R 3              // Pin LED rouge (chauffage)
#define LED_G 4              // Pin LED verte
#define LED_B 5              // Pin LED bleue
#define SERVO_PIN 2          // Pin servo (fenêtre)
#define SERVO_OPEN 180       // Position ouverte du servo
#define SERVO_CLOSE 0        // Position fermée du servo
#define SDA_PIN 18           // Pin SDA pour I2C
#define SCL_PIN 19           // Pin SCL pour I2C
#define BUTTON_PIN 10       // Pin bouton physique
#define SENSOR_INTERVAL 10000  // Intervalle lecture température (10 secondes)
#define TEMP_MIN  -10.0   // température minimale
#define TEMP_MAX   50.0   // température maximale

const char* ssid = "OnePlus NordCE 5G";      // Nom du réseau Wi-Fi
const char* password = "caca1234";          // Mot de passe Wi-Fi
const char* mqtt_server = "10.45.195.118";  // IP du broker MQTT
const char* TOPIC_CMD   = "home/appart/cmd";   // Topic commandes
const char* TOPIC_STATE = "home/appart/state"; // Topic état

// ===== OBJECTS =====
WiFiClient espClient;               // Client Wi-Fi pour MQTT
PubSubClient client(espClient);     // Client MQTT
Servo servo;                        // Objet servomoteur
Adafruit_BME280 bme;                // Objet capteur BME280

// ===== STATES =====
bool heaterState = false;           // Etat du chauffage (on/off)
bool windowState = false;           // Etat fenêtre (ouverte/fermée)
bool lastWindowState = false;       // Dernier état fenêtre (pour éviter d'envoyer le servo inutilement)
bool mqttUpdated = false;           // Flag pour savoir si un nouveau message MQTT est reçu
float temperature = 0.0;            // Température actuelle
unsigned long lastSensorSend = 0;   // Timestamp dernière lecture température
StaticJsonDocument<256> mqttDoc;    // JSON pour messages MQTT

// ===== WIFI =====
void setupWiFi() {
  WiFi.begin(ssid, password);            // Connexion au Wi-Fi
  Serial.print("Connexion WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");                   // Affiche les points tant que non connecté
  }
  Serial.println("\nWiFi connecté");     // Une fois connecté
}

// ===== MQTT =====
// Fonction appelée à chaque message MQTT reçu
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  if (deserializeJson(mqttDoc, payload, length) == DeserializationError::Ok) {
    mqttUpdated = true;                  // On met à jour le flag si JSON valide
  }
}

// Reconnexion au broker MQTT si déconnecté
void mqttReconnect() {
  if (client.connected()) return;       // Si déjà connecté, rien à faire

  while (!client.connected()) {         // Boucle jusqu'à connexion
    String id = "ESP32-" + String(random(0xffff), HEX); // ID unique
    if (client.connect(id.c_str())) {
      client.subscribe(TOPIC_CMD);      // S'abonner au topic commande
      Serial.println("MQTT connecté");
    } else {
      Serial.print("Erreur MQTT rc=");
      Serial.println(client.state());   // Affiche le code d'erreur
      delay(2000);
    }
  }
}

// ===== MQTT LOGIC =====
// Met à jour les états à partir du message MQTT reçu
void listenMqtt() {
  if (!mqttUpdated) return;             // Si pas de nouveau message, on sort

  if (mqttDoc.containsKey("heater")) heaterState = mqttDoc["heater"];  // Maj chauffage
  if (mqttDoc.containsKey("window")) windowState = mqttDoc["window"];  // Maj fenêtre

  mqttUpdated = false;                  // Reset flag
}

// ===== LED (HEATER) =====
// Met à jour la LED en fonction de l'état du chauffage
void updateLED() {
  digitalWrite(LED_R, heaterState ? HIGH : LOW); // Rouge allumé si chauffage ON
  digitalWrite(LED_G, heaterState ? LOW : HIGH); // Bleu allumé si chauffage OFF
  digitalWrite(LED_B, LOW);                       // Vert toujours éteint
}

// ===== SERVO (WINDOW) =====
// Ouvre/ferme la fenêtre seulement si l'état a changé
void updateServo() {
  if (windowState != lastWindowState) {               // Evite d'envoyer la même commande
    servo.write(windowState ? SERVO_OPEN : SERVO_CLOSE); // Position servo
    Serial.println(windowState ? "Fenêtre OUVERTE" : "Fenêtre FERMÉE");
    lastWindowState = windowState;                   // Sauvegarde nouvel état
  }
}

// ===== TEMPERATURE =====
// Lit la température depuis le capteur BME280
void readTemperature() {
  float temp = bme.readTemperature();   // Lecture température

  // Vérifie si la valeur est valide et réaliste
  if (!isnan(temp) && temp >= TEMP_MIN && temp <= TEMP_MAX) {
    temperature = temp;                 // Mise à jour seulement si cohérente
    Serial.print("Température valide = ");
    Serial.print(temperature);
    Serial.println(" °C");
  } else {
    // Valeur incohérente ignorée
    Serial.print("Température incohérente ignorée : ");
    Serial.println(temp);
  }
}

// ===== PUBLISH STATE =====
// Publie l'état courant sur MQTT
void publishState() {
  if (!client.connected()) return;                 // Vérifie connexion MQTT

  // JSON état complet
  StaticJsonDocument<256> doc;
  doc["heater"] = heaterState ? 1 : 0;
  doc["window"] = windowState ? 1 : 0;
  doc["temperature"] = temperature;

  char buffer[128];
  serializeJson(doc, buffer);                      // Sérialisation JSON
  client.publish(TOPIC_STATE, buffer);            // Envoi sur topic état

  // Envoi commande minimal pour interface 3D (Godot)
  StaticJsonDocument<64> cmdDoc;
  cmdDoc["heater"] = heaterState ? 1 : 0;
  char cmdBuffer[64];
  serializeJson(cmdDoc, cmdBuffer);
  client.publish(TOPIC_CMD, cmdBuffer, true);

  // Debug
  Serial.print("MQTT cmd envoyé : ");
  Serial.println(cmdBuffer);
  Serial.print("MQTT state envoyé : ");
  Serial.println(buffer);
}

// ===== SETUP =====
void setup() {
  Serial.begin(115200);                           // Initialise le port série
  Serial.println("ESP32 - Heater LED + Window Servo + BME280");

  // Config pins LED
  pinMode(LED_R, OUTPUT);
  pinMode(LED_G, OUTPUT);
  pinMode(LED_B, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);             // Bouton

  servo.attach(SERVO_PIN);                        // Attache servo à la pin
  servo.write(SERVO_CLOSE);                       // Position initiale fermée

  Wire.begin(SDA_PIN, SCL_PIN);                   // Initialise bus I2C

  // Test de présence du capteur BME280
  if (!bme.begin(0x76)) {
    Serial.println("BME280 pas trouvé en 0x76, test 0x77...");
    if (!bme.begin(0x77)) Serial.println("BME280 introuvable");
    else Serial.println("BME280 trouvé en 0x77");
  } else {
    Serial.println("BME280 trouvé en 0x76");
  }

  setupWiFi();                                    // Connexion Wi-Fi
  client.setServer(mqtt_server, 1883);           // Adresse du broker MQTT
  client.setCallback(mqttCallback);              // Fonction callback MQTT
}

// ===== HANDLE BUTTONS =====
// Gestion du bouton physique pour inverser le chauffage
void handleButtons() {
    static bool lastButtonState = HIGH;           // État précédent du bouton
    unsigned long now = millis();

    // Affiche debug bouton toutes les 1 sec
    static unsigned long lastDebug = 0;
    if (now - lastDebug >= 1000) {
        lastDebug = now;
        Serial.print("BUTTON: "); Serial.println(digitalRead(BUTTON_PIN));
    }

    bool currentState = digitalRead(BUTTON_PIN);

    // Détection front descendant (appui)
    if (lastButtonState == HIGH && currentState == LOW) {
        heaterState = !heaterState;               // Inverse l'état du chauffage
        Serial.println(heaterState ? "Chauffage ALLUME (bouton)" : "Chauffage ETEINT (bouton)");
        publishState();                           // Envoi MQTT immédiat
    }

    lastButtonState = currentState;               // Sauvegarde état actuel
}

// ===== LOOP =====
void loop() {
  if (!client.connected()) mqttReconnect();       // Reconnexion si nécessaire
  client.loop();                                  // MQTT loop pour recevoir messages

  listenMqtt();                                   // Met à jour états si message reçu
  handleButtons();                                // Vérifie le bouton
  updateLED();                                    // Met à jour LED chauffage
  updateServo();                                  // Met à jour servo fenêtre

  // Lire température toutes les 10 sec et publier état
  unsigned long now = millis();
  if (now - lastSensorSend >= SENSOR_INTERVAL) {
    lastSensorSend = now;
    readTemperature();                            // Lecture capteur
    publishState();                               // Envoi état MQTT
  }
}
